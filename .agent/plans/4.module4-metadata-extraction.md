# Module 4: Metadata Extraction

## Complexity: Medium — multiple coordinated files, each independently testable

## Context
Pure vector search has a ceiling. When multiple documents are uploaded, retrieval searches all chunks with no awareness of which document they belong to. Module 4 adds two LLM-powered steps:
1. **Ingestion time** — extract structured metadata (company, year, quarter, type) from each document and store as JSONB
2. **Query time** — extract filter intent from the user's question and narrow candidate documents before vector search

Result: questions about "Apple Q3 2024" only search Apple Q3 2024 chunks — not Tesla, not 2022.

## SQL — Run in Supabase SQL Editor (in order)

```sql
-- 1. Add metadata column
ALTER TABLE documents ADD COLUMN IF NOT EXISTS metadata JSONB;

-- 2. Update match_chunks RPC (backward-compatible — DEFAULT NULL preserves existing behavior)
DROP FUNCTION IF EXISTS match_chunks(vector(384), uuid, int, float);

CREATE OR REPLACE FUNCTION match_chunks(
  query_embedding     vector(384),
  match_user_id       uuid,
  match_count         int,
  match_threshold     float,
  filter_document_ids uuid[] DEFAULT NULL
)
RETURNS TABLE (id uuid, document_id uuid, content text, similarity float)
LANGUAGE plpgsql AS $$
BEGIN
  RETURN QUERY
  SELECT c.id, c.document_id, c.content,
         1 - (c.embedding <=> query_embedding) AS similarity
  FROM chunks c
  WHERE c.user_id = match_user_id
    AND 1 - (c.embedding <=> query_embedding) >= match_threshold
    AND (filter_document_ids IS NULL OR c.document_id = ANY(filter_document_ids))
  ORDER BY c.embedding <=> query_embedding
  LIMIT match_count;
END;
$$;
```

### NEW: `backend/app/services/metadata_service.py`
Uses `get_traced_client()` from `tracing.py` — same pattern as `chat_service.py`.

**Pydantic models:**
```python
class DocumentMetadata(BaseModel):
    document_type: str | None = None  # "earnings_report", "whitepaper", "contract"
    topic: str | None = None
    entity: str | None = None         # company or organization name
    year: int | None = None
    quarter: str | None = None        # "Q1" / "Q2" / "Q3" / "Q4"
    language: str | None = None       # ISO 639-1 e.g. "en"
    summary: str | None = None        # 1-2 sentence summary

class QueryFilters(BaseModel):
    entity: str | None = None
    year: int | None = None
    quarter: str | None = None
    document_type: str | None = None
```

**`extract_document_metadata(text: str) -> DocumentMetadata`**
- Sends `text[:3000]` to Groq with `response_format={"type": "json_object"}`, `max_tokens=400`, `temperature=0`
- Parses response with `DocumentMetadata.model_validate_json()`
- On ANY exception: log and return `DocumentMetadata()` — never raise

**`extract_query_filters(query: str) -> QueryFilters`**
- Same pattern, `max_tokens=150` (tight — keeps chat latency low)
- On ANY exception: return `QueryFilters()` — fails open, unfiltered search runs

Update `retrieve_context(user_id, query)` — add pre-filter step before embedding:
```python
# 1. Extract query filters (fails open → QueryFilters() with all None)
filters = await extract_query_filters(query)
filter_document_ids = None

has_filters = any(v is not None for v in [filters.entity, filters.year, filters.quarter, filters.document_type])

if has_filters:
    jsonb_filter = {k: v for k, v in filters.model_dump().items() if v is not None}

    doc_result = db.table("documents").select("id").eq("user_id", user_id).contains("metadata", jsonb_filter).execute()
    matched_ids = [r["id"] for r in (doc_result.data or [])]

    if matched_ids:
        filter_document_ids = matched_ids
    # else: CRITICAL FALLBACK — zero matches → run unfiltered (filter_document_ids stays None)

# 2. Embed query (unchanged)
# 3. Call match_chunks RPC — add filter_document_ids to params only if not None
rpc_params = {"query_embedding": ..., "match_user_id": user_id, "match_count": TOP_K, "match_threshold": SIMILARITY_THRESHOLD}
if filter_document_ids is not None:
    rpc_params["filter_document_ids"] = filter_document_ids
```

## Tests

### Why write tests for this module?
`metadata_service.py` is the riskiest new code — it calls an external LLM and must **never crash ingestion or retrieval** even when things go wrong. Tests verify both the happy path (correct Pydantic parsing) and the failure path (graceful fallback) without making real API calls.

### Setup — add to `backend/requirements.txt`
```
pytest==8.3.2
pytest-asyncio==0.24.0
```
Install: `cd backend && venv/Scripts/pip install pytest==8.3.2 pytest-asyncio==0.24.0`

```python
import pytest
from unittest.mock import AsyncMock, MagicMock, patch
from app.services.metadata_service import (
    extract_document_metadata,
    extract_query_filters,
    DocumentMetadata,
    QueryFilters,
)

pytestmark = pytest.mark.asyncio


def _mock_llm_response(json_str: str):
    """Helper: build a fake Groq response object returning the given JSON string."""
    response = MagicMock()
    response.choices[0].message.content = json_str
    return response


# extract_document_metadata

async def test_metadata_parses_all_fields():
    """LLM returns well-formed JSON → all Pydantic fields populated correctly."""
    fake_json = '{"entity":"Apple","year":2024,"quarter":"Q3","document_type":"earnings_report","language":"en","topic":"quarterly results","summary":"Apple Q3 2024 earnings."}'
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response(fake_json)
        )
        result = await extract_document_metadata("Apple Q3 2024 earnings...")

    assert result.entity == "Apple"
    assert result.year == 2024
    assert result.quarter == "Q3"
    assert result.document_type == "earnings_report"
    assert result.language == "en"


async def test_metadata_year_coerced_from_string():
    """LLM returns year as a string → Pydantic coerces it to int."""
    fake_json = '{"entity":"Tesla","year":"2023"}'
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response(fake_json)
        )
        result = await extract_document_metadata("some text")

    assert result.year == 2023
    assert isinstance(result.year, int)


async def test_metadata_partial_fields():
    """LLM returns only some fields → missing fields are None, not an error."""
    fake_json = '{"entity":"Microsoft"}'
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response(fake_json)
        )
        result = await extract_document_metadata("some text")

    assert result.entity == "Microsoft"
    assert result.year is None
    assert result.quarter is None


async def test_metadata_llm_error_returns_empty():
    """LLM call raises an exception → returns empty DocumentMetadata, never raises."""
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            side_effect=Exception("API timeout")
        )
        result = await extract_document_metadata("some text")

    assert isinstance(result, DocumentMetadata)
    assert result.entity is None
    assert result.year is None


async def test_metadata_invalid_json_returns_empty():
    """LLM returns malformed JSON → Pydantic parse fails → returns empty DocumentMetadata."""
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response("not valid json {{")
        )
        result = await extract_document_metadata("some text")

    assert isinstance(result, DocumentMetadata)
    assert result.entity is None


# extract_query_filters

async def test_filters_parses_correctly():
    """LLM extracts entity, year, and quarter from a specific question."""
    fake_json = '{"entity":"Tesla","year":2023,"quarter":"Q2"}'
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response(fake_json)
        )
        result = await extract_query_filters("What were Tesla's Q2 2023 results?")

    assert result.entity == "Tesla"
    assert result.year == 2023
    assert result.quarter == "Q2"


async def test_filters_empty_for_general_question():
    """General question → LLM returns empty JSON → all filters are None."""
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            return_value=_mock_llm_response("{}")
        )
        result = await extract_query_filters("Summarize the main points")

    assert result.entity is None
    assert result.year is None
    assert result.quarter is None
    assert result.document_type is None


async def test_filters_llm_error_returns_empty():
    """LLM call fails → returns empty QueryFilters, never raises."""
    with patch("app.services.metadata_service.get_traced_client") as mock_fn:
        mock_fn.return_value.chat.completions.create = AsyncMock(
            side_effect=Exception("rate limit")
        )
        result = await extract_query_filters("What were Apple's results?")

    assert isinstance(result, QueryFilters)
    assert result.entity is None
```
